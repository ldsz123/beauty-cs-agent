## 二、三级处理流水线详细设计

### 2.1 第一级：情感识别层

**目标**：识别用户情绪状态，为后续处理提供情感上下文

```python
# emotion_layer.py
from enum import Enum
from typing import Optional
from dataclasses import dataclass

class EmotionType(Enum):
    """电商场景情感分类"""
    ANGRY = "angry"           # 愤怒
    ANXIOUS = "anxious"       # 焦虑
    DISSATISFIED = "dissatisfied"  # 不满
    NEUTRAL = "neutral"       # 中性
    SATISFIED = "satisfied"   # 满意
    URGENT = "urgent"         # 紧急
    GRATEFUL = "grateful"     # 感激

class EmotionIntensity(Enum):
    """情感强度"""
    LOW = 1
    MEDIUM = 2
    HIGH = 3
    EXTREME = 4

@dataclass
class EmotionResult:
    emotion: EmotionType
    intensity: EmotionIntensity
    confidence: float
    triggers: list[str]  # 触发情感的关键词/短语
    raw_score: dict     # 各情感维度原始分数

class EmotionRecognitionLayer:
    """
    情感识别层
    - 基于 LLM + 规则引擎双路识别
    - 支持中文电商场景的情感分类
    - 输出情感类型、强度、置信度
    """
    
    def __init__(self, llm_client, emotion_lexicon_path: str):
        self.llm = llm_client
        self.lexicon = self._load_emotion_lexicon(emotion_lexicon_path)
        
        # 电商场景情感关键词库
        self.emotion_keywords = {
            EmotionType.ANGRY: [
                "垃圾", "骗子", "投诉", "差评", "退款", "坑人",
                "太差", "骗钱", "虚假宣传", "态度恶劣", "再也不买"
            ],
            EmotionType.ANXIOUS: [
                "怎么还没", "什么时候", "急", "担心", "会不会",
                "能不能", "怎么办", "来不及", "快点"
            ],
            EmotionType.URGENT: [
                "紧急", "加急", "马上", "立刻", "今天必须",
                "等不及", "着急", "赶时间"
            ],
            EmotionType.DISSATISFIED: [
                "不满意", "不太行", "一般般", "凑合", "勉强",
                "不如", "没有想象中", "差强人意"
            ],
            EmotionType.SATISFIED: [
                "不错", "满意", "好评", "喜欢", "推荐", "物超所值",
                "质量好", "速度快", "很棒"
            ],
        }
    
    async def analyze(self, user_message: str, 
                      conversation_history: list[dict] = None) -> EmotionResult:
        """
        双路情感识别：
        1. 规则引擎：基于关键词匹配（快速、低延迟）
        2. LLM 推理：基于上下文理解（准确、语义级）
        最终融合两路结果
        """
        # 规则引擎快速识别
        rule_result = self._rule_based_analysis(user_message)
        
        # LLM 深度分析
        llm_result = await self._llm_analysis(user_message, conversation_history)
        
        # 融合两路结果
        return self._fuse_results(rule_result, llm_result)
    
    def _rule_based_analysis(self, message: str) -> EmotionResult:
        """基于关键词的规则匹配"""
        scores = {emotion: 0.0 for emotion in EmotionType}
        triggers = []
        
        for emotion, keywords in self.emotion_keywords.items():
            for keyword in keywords:
                if keyword in message:
                    scores[emotion] += 1.0
                    triggers.append(keyword)
        
        # 归一化
        total = sum(scores.values()) or 1.0
        scores = {k: v/total for k, v in scores.items()}
        
        # 取最高分的情感
        max_emotion = max(scores, key=scores.get)
        max_score = scores[max_emotion]
        
        # 映射强度
        if max_score > 0.6:
            intensity = EmotionIntensity.HIGH
        elif max_score > 0.3:
            intensity = EmotionIntensity.MEDIUM
        else:
            intensity = EmotionIntensity.LOW
            
        return EmotionResult(
            emotion=max_emotion,
            intensity=intensity,
            confidence=max_score,
            triggers=triggers,
            raw_score={e.value: s for e, s in scores.items()}
        )
    
    async def _llm_analysis(self, message: str, 
                            history: list[dict] = None) -> EmotionResult:
        """基于 LLM 的情感分析"""
        prompt = f"""你是一个电商客服情感分析专家。请分析以下用户消息的情感状态。

用户消息：{message}

请从以下维度分析：
1. 主要情感类型（愤怒/焦虑/不满/中性/满意/紧急/感激）
2. 情感强度（1-4级，4为最强烈）
3. 触发情感的关键短语
4. 是否需要人工介入（当情感强度≥3且为负面情感时）

以JSON格式返回：
{{"emotion": "类型", "intensity": 级别, "confidence": 0-1, "triggers": ["短语1", "短语2"], "need_human": true/false}}"""

        response = await self.llm.chat(prompt, response_format="json")
        return EmotionResult(
            emotion=EmotionType(response["emotion"]),
            intensity=EmotionIntensity(response["intensity"]),
            confidence=response["confidence"],
            triggers=response["triggers"],
            raw_score={}
        )
    
    def _fuse_results(self, rule_result: EmotionResult, 
                      llm_result: EmotionResult) -> EmotionResult:
        """融合规则引擎和 LLM 的结果"""
        # LLM 权重更高（0.7），规则引擎作为补充（0.3）
        if rule_result.emotion == llm_result.emotion:
            # 一致时提升置信度
            confidence = min(0.95, 
                           rule_result.confidence * 0.3 + llm_result.confidence * 0.7)
            return EmotionResult(
                emotion=llm_result.emotion,
                intensity=llm_result.intensity,
                confidence=confidence,
                triggers=list(set(rule_result.triggers + llm_result.triggers)),
                raw_score=rule_result.raw_score
            )
        else:
            # 不一致时以 LLM 为主
            return llm_result
```

### 2.2 第二级：语义分类层

**目标**：理解用户意图，将用户消息分类到具体的业务场景

```python
# semantic_layer.py
from enum import Enum
from dataclasses import dataclass, field

class IntentType(Enum):
    """电商客服意图分类"""
    # 售前咨询
    PRODUCT_INQUIRY = "product_inquiry"       # 商品咨询
    PRICE_INQUIRY = "price_inquiry"           # 价格咨询
    PROMOTION_INQUIRY = "promotion_inquiry"   # 促销活动咨询
    COMPARISON = "comparison"                 # 商品对比
    
    # 售中服务
    ORDER_STATUS = "order_status"             # 订单查询
    ORDER_MODIFY = "order_modify"             # 订单修改
    PAYMENT_ISSUE = "payment_issue"           # 支付问题
    SHIPPING_INQUIRY = "shipping_inquiry"     # 物流查询
    
    # 售后服务
    RETURN_REFUND = "return_refund"           # 退货退款
    EXCHANGE = "exchange"                     # 换货
    COMPLAINT = "complaint"                   # 投诉
    WARRANTY = "warranty"                     # 保修
    
    # 其他
    ACCOUNT_ISSUE = "account_issue"           # 账户问题
    TECH_SUPPORT = "tech_support"             # 技术支持
    GENERAL_CHAT = "general_chat"             # 闲聊
    ESCALATE = "escalate"                     # 转人工

class UrgencyLevel(Enum):
    LOW = "low"
    MEDIUM = "medium"
    HIGH = "high"
    CRITICAL = "critical"

@dataclass
class SemanticResult:
    primary_intent: IntentType          # 主意图
    secondary_intents: list[IntentType] # 次要意图
    confidence: float                   # 置信度
    urgency: UrgencyLevel              # 紧急程度
    context_dependencies: list[str]     # 上下文依赖
    required_info: list[str]           # 需要补充的信息

class SemanticClassificationLayer:
    """
    语义分类层
    - 基于 LLM 的意图识别
    - 支持多意图识别
    - 结合情感上下文调整分类
    """
    
    def __init__(self, llm_client, intent_examples_path: str):
        self.llm = llm_client
        self.intent_examples = self._load_intent_examples(intent_examples_path)
    
    async def classify(self, user_message: str,
                       emotion_result: EmotionResult = None,
                       conversation_context: dict = None) -> SemanticResult:
        """
        语义分类主流程
        """
        # 构建上下文增强的 prompt
        context_info = ""
        if emotion_result:
            context_info = f"\n用户当前情感状态：{emotion_result.emotion.value}（强度：{emotion_result.intensity.value}）"
        
        prompt = f"""你是电商客服意图识别专家。请分析用户消息的意图。

用户消息：{user_message}
{context_info}

意图类别：
- 售前：product_inquiry(商品咨询), price_inquiry(价格), promotion_inquiry(促销), comparison(对比)
- 售中：order_status(订单查询), order_modify(订单修改), payment_issue(支付问题), shipping_inquiry(物流)
- 售后：return_refund(退货退款), exchange(换货), complaint(投诉), warranty(保修)
- 其他：account_issue(账户), tech_support(技术支持), general_chat(闲聊), escalate(转人工)

请返回JSON：
{{
  "primary_intent": "主意图",
  "secondary_intents": ["次要意图1", "次要意图2"],
  "confidence": 0.0-1.0,
  "urgency": "low/medium/high/critical",
  "context_dependencies": ["依赖的上下文信息"],
  "required_info": ["需要向用户确认的信息"],
  "reasoning": "分类理由"
}}"""

        response = await self.llm.chat(prompt, response_format="json")
        
        # 情感上下文修正
        urgency = self._adjust_urgency(
            UrgencyLevel(response["urgency"]), 
            emotion_result
        )
        
        return SemanticResult(
            primary_intent=IntentType(response["primary_intent"]),
            secondary_intents=[IntentType(i) for i in response.get("secondary_intents", [])],
            confidence=response["confidence"],
            urgency=urgency,
            context_dependencies=response.get("context_dependencies", []),
            required_info=response.get("required_info", [])
        )
    
    def _adjust_urgency(self, base_urgency: UrgencyLevel,
                        emotion: EmotionResult) -> UrgencyLevel:
        """根据情感状态调整紧急程度"""
        if emotion and emotion.emotion in [EmotionType.ANGRY, EmotionType.URGENT]:
            if emotion.intensity.value >= 3:
                return UrgencyLevel.CRITICAL
            elif emotion.intensity.value >= 2:
                return UrgencyLevel.HIGH
        return base_urgency
```

### 2.3 第三级：实体提取层

**目标**：从用户消息中提取关键业务实体信息

```python
# entity_layer.py
from dataclasses import dataclass, field
from typing import Optional
from datetime import datetime

@dataclass
class ExtractedEntities:
    """提取的实体集合"""
    # 订单相关
    order_id: Optional[str] = None           # 订单号
    tracking_number: Optional[str] = None    # 物流单号
    
    # 商品相关
    product_name: Optional[str] = None       # 商品名称
    product_sku: Optional[str] = None        # 商品SKU
    product_category: Optional[str] = None   # 商品类目
    
    # 用户信息
    phone: Optional[str] = None              # 手机号
    address: Optional[str] = None            # 地址
    user_id: Optional[str] = None            # 用户ID
    
    # 金额相关
    amount: Optional[float] = None           # 金额
    quantity: Optional[int] = None           # 数量
    
    # 时间相关
    time_reference: Optional[str] = None     # 时间引用
    date: Optional[str] = None               # 日期
    
    # 问题描述
    issue_type: Optional[str] = None         # 问题类型
    issue_description: Optional[str] = None  # 问题描述
    
    # 图片信息
    has_image: bool = False                  # 是否包含图片
    image_description: Optional[str] = None  # 图片描述
    
    # 原始提取结果
    raw_entities: dict = field(default_factory=dict)
    confidence: float = 0.0

class EntityExtractionLayer:
    """
    实体提取层
    - 基于 LLM + 正则表达式双路提取
    - 支持订单号、手机号等固定格式实体的精确提取
    - 支持商品名、问题描述等模糊实体的语义提取
    """
    
    def __init__(self, llm_client):
        self.llm = llm_client
        
        # 正则规则（精确匹配）
        self.patterns = {
            "order_id": [
                r"订单[号编号]?[：:]?\s*([A-Z0-9]{10,20})",
                r"([A-Z]{2}\d{12,18})",  # 如 TB20240101123456
                r"(\d{16,20})",           # 纯数字订单号
            ],
            "phone": [
                r"1[3-9]\d{9}",
            ],
            "tracking_number": [
                r"物流[号单]?[：:]?\s*([A-Z0-9]{10,25})",
                r"([A-Z]{2}\d{9}[A-Z]{2})",  # 国际物流
            ],
            "amount": [
                r"[￥¥$]?(\d+\.?\d*)\s*元?",
            ],
        }
    
    async def extract(self, user_message: str,
                      intent: IntentType = None,
                      image_info: dict = None) -> ExtractedEntities:
        """
        实体提取主流程
        """
        # 1. 正则精确提取
        regex_entities = self._regex_extract(user_message)
        
        # 2. LLM 语义提取
        llm_entities = await self._llm_extract(user_message, intent)
        
        # 3. 合并结果（LLM 结果优先，正则作为补充/校验）
        merged = self._merge_entities(regex_entities, llm_entities)
        
        # 4. 图片实体提取
        if image_info:
            merged.has_image = True
            merged.image_description = image_info.get("description")
        
        return merged
    
    def _regex_extract(self, message: str) -> dict:
        """正则表达式精确提取"""
        import re
        results = {}
        for entity_type, patterns in self.patterns.items():
            for pattern in patterns:
                match = re.search(pattern, message)
                if match:
                    results[entity_type] = match.group(1) if match.groups() else match.group(0)
                    break
        return results
    
    async def _llm_extract(self, message: str, 
                           intent: IntentType = None) -> dict:
        """LLM 语义提取"""
        intent_hint = f"当前意图：{intent.value}" if intent else ""
        
        prompt = f"""你是电商客服实体提取专家。请从用户消息中提取所有业务相关实体。

用户消息：{message}
{intent_hint}

需要提取的实体类型：
- order_id: 订单号
- tracking_number: 物流单号
- product_name: 商品名称
- product_sku: 商品SKU
- phone: 手机号
- address: 地址
- amount: 金额（数字）
- quantity: 数量（数字）
- time_reference: 时间引用
- issue_type: 问题类型
- issue_description: 问题描述

以JSON格式返回提取结果，未找到的字段返回null。"""

        response = await self.llm.chat(prompt, response_format="json")
        return response
    
    def _merge_entities(self, regex_result: dict, llm_result: dict) -> ExtractedEntities:
        """合并两路提取结果"""
        # 合并逻辑：正则结果作为 LLM 结果的补充和校验
        merged = {**llm_result, **{k: v for k, v in regex_result.items() if k not in llm_result or llm_result[k] is None}}
        
        return ExtractedEntities(
            order_id=merged.get("order_id"),
            tracking_number=merged.get("tracking_number"),
            product_name=merged.get("product_name"),
            product_sku=merged.get("product_sku"),
            phone=merged.get("phone"),
            address=merged.get("address"),
            amount=merged.get("amount"),
            quantity=merged.get("quantity"),
            time_reference=merged.get("time_reference"),
            issue_type=merged.get("issue_type"),
            issue_description=merged.get("issue_description"),
            raw_entities=merged,
            confidence=0.85 if merged else 0.0
        )
```

```python
from dataclasses import dataclass

@dataclass
class ImageUnderstandingResult:

    async def understand(self, image_data: bytes, 
                         context: str = "") -> ImageUnderstandingResult:
        """
        图片理解主流程

        prompt = f"""你是一个电商客服图片理解专家。请分析这张图片。

上下文：{context}

请分析：
1. 图片类型（商品截图/物流截图/聊天记录/支付凭证/商品问题照片等）
2. 图片内容详细描述
3. 图片中可见的文字信息
4. 与用户问题相关的关键信息

以JSON格式返回：
{{
  "image_type": "类型",
  "description": "详细描述",
  "extracted_text": "图片中的文字",
  "relevant_info": {{"key": "value"}},
  "confidence": 0.0-1.0
}}"""

        result = await self.vision_llm.chat_with_image(prompt, image_data, response_format="json")

            description=result["description"],
            image_type=result["image_type"],
            extracted_text=result.get("extracted_text", ""),
            relevant_info=result.get("relevant_info", {}),
            confidence=result["confidence"]
        )
```

## 四、核心 Agent 编排引擎

```python
# agent_engine.py
from dataclasses import dataclass
from typing import Optional, AsyncGenerator
import time

@dataclass
class AgentResponse:
    """Agent 最终响应"""
    reply: str                      # 回复内容
    emotion_result: EmotionResult   # 情感分析结果
    semantic_result: SemanticResult # 语义分类结果
    entities: ExtractedEntities     # 实体提取结果
    actions: list[dict]             # 执行的业务动作
    need_human: bool               # 是否需要转人工
    confidence: float              # 整体置信度
    processing_time_ms: int        # 处理耗时

class CustomerServiceAgent:
    """
    电商智能客服 Agent 核心引擎
    编排三级处理流水线，实现深度理解
    """
    
    def __init__(self, config: dict):
        # 初始化各层
        self.emotion_layer = EmotionRecognitionLayer(
            llm_client=config["llm_client"],
            emotion_lexicon_path=config.get("emotion_lexicon", "data/emotion_lexicon.json")
        )
        self.semantic_layer = SemanticClassificationLayer(
            llm_client=config["llm_client"],
            intent_examples_path=config.get("intent_examples", "data/intent_examples.json")
        )
        self.entity_layer = EntityExtractionLayer(
            llm_client=config["llm_client"]
        )
        self.vision_module = ImageUnderstandingModule(
            vision_llm_client=config["vision_llm_client"]
        )
        
        # 业务执行器
        self.action_executor = BusinessActionExecutor(config)
        
        # 知识库
        self.knowledge_base = KnowledgeBase(config.get("kb_config", {}))
        
        # 会话管理
        self.session_manager = SessionManager()
    
    async def process(self, user_message: str,
                      session_id: str,
                      image_data: bytes = None) -> AgentResponse:
        """
        主处理流程 — 三级流水线
        """
        start_time = time.time()
        
        # 获取会话上下文
        session = await self.session_manager.get_session(session_id)
        
        # ========== 第一级：情感识别 ==========
        emotion_result = await self.emotion_layer.analyze(
            user_message, 
            conversation_history=session.get("history", [])
        )
        
        # ========== 第二级：语义分类 ==========
        semantic_result = await self.semantic_layer.classify(
            user_message,
            emotion_result=emotion_result,
            conversation_context=session.get("context", {})
        )
        
        # ========== 第三级：实体提取 ==========
        image_info = None
        if image_data:
            image_info = await self.vision_module.understand(
                image_data, 
                context=user_message
            )
        
        entities = await self.entity_layer.extract(
            user_message,
            intent=semantic_result.primary_intent,
            image_info=image_info.__dict__ if image_info else None
        )
        
        # ========== 业务动作执行 ==========
        actions = await self.action_executor.execute(
            intent=semantic_result.primary_intent,
            entities=entities,
            session=session
        )
        
        # ========== 知识库检索 ==========
        kb_context = await self.knowledge_base.search(
            query=user_message,
            intent=semantic_result.primary_intent,
            entities=entities
        )
        
        # ========== 生成回复 ==========
        reply = await self._generate_response(
            user_message=user_message,
            emotion=emotion_result,
            semantic=semantic_result,
            entities=entities,
            actions=actions,
            kb_context=kb_context,
            session=session
        )
        
        # ========== 判断是否需要转人工 ==========
        need_human = self._check_escalation(
            emotion_result, semantic_result, session
        )
        
        # 更新会话
        await self.session_manager.update_session(
            session_id, 
            user_message, 
            reply,
            emotion_result,
            semantic_result
        )
        
        processing_time = int((time.time() - start_time) * 1000)
        
        return AgentResponse(
            reply=reply,
            emotion_result=emotion_result,
            semantic_result=semantic_result,
            entities=entities,
            actions=actions,
            need_human=need_human,
            confidence=semantic_result.confidence,
            processing_time_ms=processing_time
        )
    
    async def _generate_response(self, user_message, emotion, semantic, 
                                  entities, actions, kb_context, session) -> str:
        """生成最终回复"""
        
        # 根据情感调整回复风格
        tone_instruction = self._get_tone_instruction(emotion)
        
        # 构建上下文
        context = f"""知识库信息：{kb_context}
已执行的操作：{actions}
提取的订单信息：订单号 {entities.order_id or '未知'}"""

        prompt = f"""你是电商客服助手。请根据以下信息生成专业、友好的回复。

{tone_instruction}

用户消息：{user_message}
意图：{semantic.primary_intent.value}
{context}

回复要求：
1. 先回应用户的情感状态（如道歉、感谢等）
2. 针对用户意图给出具体解决方案
3. 如需用户补充信息，清晰列出
4. 语气专业、友好、简洁
5. 中文回复

请直接输出回复内容。"""

        response = await self.llm.chat(prompt)
        return response
    
    def _get_tone_instruction(self, emotion: EmotionResult) -> str:
        """根据情感状态生成语气指导"""
        tone_map = {
            EmotionType.ANGRY: "用户非常愤怒，请先真诚道歉，表达理解和重视，然后快速给出解决方案。语气要诚恳、耐心。",
            EmotionType.ANXIOUS: "用户感到焦虑，请先安抚情绪，表达会尽快处理，给出明确的时间承诺。",
            EmotionType.URGENT: "用户很着急，请优先处理，给出最快的解决方案和时间节点。",
            EmotionType.DISSATISFIED: "用户不满意，先承认不足，给出补偿或改进方案。",
            EmotionType.SATISFIED: "用户满意，感谢认可，保持友好。",
            EmotionType.GRATEFUL: "用户表示感谢，礼貌回应。",
            EmotionType.NEUTRAL: "正常客服语气，专业友好。",
        }
        return tone_map.get(emotion.emotion, "正常客服语气。")
    
    def _check_escalation(self, emotion: EmotionResult, 
                          semantic: SemanticResult,
                          session: dict) -> bool:
        """判断是否需要转人工"""
        # 触发转人工的条件
        if emotion.emotion == EmotionType.ANGRY and emotion.intensity.value >= 3:
            return True
        if semantic.urgency == UrgencyLevel.CRITICAL:
            return True
        # 连续3轮未解决
        if session.get("unresolved_count", 0) >= 3:
            return True
        # 用户明确要求人工
        if semantic.primary_intent == IntentType.ESCALATE:
            return True
        return False
```

```python
# action_executor.py
class BusinessActionExecutor:
    """
    业务动作执行器
    根据意图和实体执行具体的业务操作
    """
    
    def __init__(self, config: dict):
        self.order_service = OrderService(config.get("order_api"))
        self.logistics_service = LogisticsService(config.get("logistics_api"))
        self.product_service = ProductService(config.get("product_api"))
        self.refund_service = RefundService(config.get("refund_api"))
        self.user_service = UserService(config.get("user_api"))
    
    async def execute(self, intent: IntentType, 
                      entities: ExtractedEntities,
                      session: dict) -> list[dict]:
        """根据意图执行对应业务动作"""
        
        action_map = {
            IntentType.ORDER_STATUS: self._query_order,
            IntentType.SHIPPING_INQUIRY: self._query_logistics,
            IntentType.RETURN_REFUND: self._initiate_return,
            IntentType.PRODUCT_INQUIRY: self._query_product,
            IntentType.PRICE_INQUIRY: self._query_price,
            IntentType.PAYMENT_ISSUE: self._handle_payment_issue,
            IntentType.ORDER_MODIFY: self._modify_order,
            IntentType.EXCHANGE: self._initiate_exchange,
            IntentType.COMPLAINT: self._handle_complaint,
            IntentType.ACCOUNT_ISSUE: self._handle_account_issue,
        }
        
        handler = action_map.get(intent)
        if handler:
            return await handler(entities, session)
        return []
    
    async def _query_order(self, entities: ExtractedEntities, session: dict):
        """查询订单状态"""
        if entities.order_id:
            order_info = await self.order_service.get_order(entities.order_id)
            return [{"action": "query_order", "result": order_info}]
        
        # 尝试从会话中获取订单号
        if session.get("last_order_id"):
            order_info = await self.order_service.get_order(session["last_order_id"])
            return [{"action": "query_order", "result": order_info}]
        
        return [{"action": "query_order", "status": "need_order_id"}]
    
    async def _query_logistics(self, entities: ExtractedEntities, session: dict):
        """查询物流信息"""
        tracking = entities.tracking_number or entities.order_id
        if tracking:
            logistics_info = await self.logistics_service.track(tracking)
            return [{"action": "query_logistics", "result": logistics_info}]
        return [{"action": "query_logistics", "status": "need_tracking_number"}]
    
    async def _initiate_return(self, entities: ExtractedEntities, session: dict):
        """发起退货退款"""
        if entities.order_id:
            result = await self.refund_service.create_refund(
                order_id=entities.order_id,
                reason=entities.issue_description or "用户申请退货"
            )
            return [{"action": "initiate_return", "result": result}]
        return [{"action": "initiate_return", "status": "need_order_id"}]
    
    async def _query_product(self, entities: ExtractedEntities, session: dict):
        """查询商品信息"""
        if entities.product_name or entities.product_sku:
            product_info = await self.product_service.search(
                name=entities.product_name,
                sku=entities.product_sku
            )
            return [{"action": "query_product", "result": product_info}]
        return [{"action": "query_product", "status": "need_product_info"}]
    
    async def _query_price(self, entities: ExtractedEntities, session: dict):
        """查询价格"""
        if entities.product_name or entities.product_sku:
            price_info = await self.product_service.get_price(
                name=entities.product_name,
                sku=entities.product_sku
            )
            return [{"action": "query_price", "result": price_info}]
        return [{"action": "query_price", "status": "need_product_info"}]
    
    async def _handle_payment_issue(self, entities: ExtractedEntities, session: dict):
        """处理支付问题"""
        return [{"action": "handle_payment_issue", "status": "processing"}]
    
    async def _modify_order(self, entities: ExtractedEntities, session: dict):
        """修改订单"""
        if entities.order_id:
            return [{"action": "modify_order", "order_id": entities.order_id, "status": "pending_confirmation"}]
        return [{"action": "modify_order", "status": "need_order_id"}]
    
    async def _initiate_exchange(self, entities: ExtractedEntities, session: dict):
        """发起换货"""
        if entities.order_id:
            result = await self.refund_service.create_exchange(
                order_id=entities.order_id,
                reason=entities.issue_description or "用户申请换货"
            )
            return [{"action": "initiate_exchange", "result": result}]
        return [{"action": "initiate_exchange", "status": "need_order_id"}]
    
    async def _handle_complaint(self, entities: ExtractedEntities, session: dict):
        """处理投诉"""
        result = await self.refund_service.create_complaint(
            order_id=entities.order_id,
            description=entities.issue_description,
            user_phone=entities.phone
        )
        return [{"action": "handle_complaint", "result": result}]
    
    async def _handle_account_issue(self, entities: ExtractedEntities, session: dict):
        """处理账户问题"""
        return [{"action": "handle_account_issue", "status": "processing"}]
```

```python
# knowledge_base.py
from typing import Optional

class KnowledgeBase:
    """
    电商知识库
    - 商品信息
    - 售后政策
    - 物流政策
    - 常见问题 FAQ
    - 促销活动
    """
    
    def __init__(self, config: dict):
        self.vector_db = MilvusClient(config.get("milvus", {}))
        self.embedding_model = EmbeddingModel(config.get("embedding", {}))
        self.cache = RedisClient(config.get("redis", {}))
    
    async def search(self, query: str, 
                     intent: IntentType = None,
                     entities: ExtractedEntities = None,
                     top_k: int = 5) -> str:
        """
        知识库检索
        """
        # 1. 检查缓存
        cache_key = f"kb:{hash(query)}"
        cached = await self.cache.get(cache_key)
        if cached:
            return cached
        
        # 2. 构建增强查询
        enhanced_query = self._build_enhanced_query(query, intent, entities)
        
        # 3. 向量检索
        query_embedding = await self.embedding_model.encode(enhanced_query)
        results = await self.vector_db.search(
            collection="ecommerce_kb",
            vector=query_embedding,
            top_k=top_k
        )
        
        # 4. 格式化结果
        context = self._format_results(results)
        
        # 5. 缓存结果
        await self.cache.set(cache_key, context, ttl=300)
        
        return context
    
    def _build_enhanced_query(self, query, intent, entities) -> str:
        """构建增强查询"""
        parts = [query]
        if intent:
            parts.append(f"意图：{intent.value}")
        if entities:
            if entities.product_name:
                parts.append(f"商品：{entities.product_name}")
            if entities.issue_type:
                parts.append(f"问题：{entities.issue_type}")
        return " | ".join(parts)
    
    def _format_results(self, results: list[dict]) -> str:
        """格式化检索结果"""
        if not results:
            return ""
        return "\n\n".join([
            f"[{r.get('category', '通用')}] {r['content']}"
            for r in results
        ])
```

```python
from fastapi import FastAPI, UploadFile, File, Form

from pydantic import BaseModel

class ChatRequest(BaseModel):
    message: str
    session_id: str
    user_id: Optional[str] = None

class ChatResponse(BaseModel):
    reply: str
    emotion: dict
    intent: str
    entities: dict
    need_human: bool
    processing_time_ms: int

@app.post("/api/chat", response_model=ChatResponse)
async def chat(request: ChatRequest):
    """同步对话接口"""
    response = await agent.process(
        user_message=request.message,
        session_id=request.session_id
    )
    return ChatResponse(
        reply=response.reply,
        emotion={
            "type": response.emotion_result.emotion.value,
            "intensity": response.emotion_result.intensity.value,
            "confidence": response.emotion_result.confidence
        },
        intent=response.semantic_result.primary_intent.value,
        entities=response.entities.__dict__,
        need_human=response.need_human,
        processing_time_ms=response.processing_time_ms
    )

@app.post("/api/chat/with-image")
async def chat_with_image(
    message: str = Form(...),
    session_id: str = Form(...),
    image: UploadFile = File(...)
):
    """带图片的对话接口"""
    image_data = await image.read()
    response = await agent.process(
        user_message=request.message,
        session_id=request.session_id,
        image_data=image_data
    )
    return {"reply": response.reply, "need_human": response.need_human}

@app.post("/api/chat/stream")
async def chat_stream(request: ChatRequest):
    """流式对话接口（SSE）"""
    async def event_generator():
        async for chunk in agent.process_stream(
            user_message=request.message,
            session_id=request.session_id
        ):
            yield f"data: {json.dumps(chunk, ensure_ascii=False)}\n\n"
    
    return StreamingResponse(event_generator(), media_type="text/event-stream")

@app.get("/api/session/{session_id}")
async def get_session(session_id: str):
    """获取会话信息"""
    session = await agent.session_manager.get_session(session_id)
    return session

@app.get("/api/health")
async def health():
    return {"status": "healthy", "version": "1.0.0"}
```
